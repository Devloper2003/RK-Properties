    18→function PropertyImage({ src, alt }: { src: string; alt: string }) {
    19→  const [loaded, setLoaded] = useState(false);
    20→  return (
    21→    <div className="relative h-64 overflow-hidden bg-gold-100 dark:bg-gray-800">
    22→      {!loaded && <div className="absolute inset-0 skeleton-shimmer" />}
    23→      <img
    24→        src={src}
    25→        alt={alt}
    26→        referrerPolicy="no-referrer"
    27→        loading="lazy"
    28→        onLoad={() => setLoaded(true)}
    29→        className={`w-full h-full object-cover transition-opacity duration-500 ${loaded ? 'opacity-100' : 'opacity-0'}`}
    30→      />
    31→    </div>
    32→  );
    33→}
